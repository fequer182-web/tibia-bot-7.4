cmake_minimum_required(VERSION 3.20)
project(CromeClassicFeatureProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(CromeClassicFeatureProbe SHARED src/FeatureProbe.cpp)
set_target_properties(CromeClassicFeatureProbe PROPERTIES PREFIX "" OUTPUT_NAME "CromeClassicFeatureProbe")
add_executable(CromeClassicProbeInjector src/Injector.cpp)
if(MINGW)
  target_link_options(CromeClassicFeatureProbe PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(CromeClassicProbeInjector PRIVATE -static-libgcc -static-libstdc++)
endif()
