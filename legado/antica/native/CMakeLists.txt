cmake_minimum_required(VERSION 3.20)
project(CrominNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Cromin requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(SOURCES rat_attack_once corpse_open_once loot_once autowalk_once talk_once look_watch item_use_once)
set(TARGETS OTC_Rat_Attack_Once OTC_Corpse_Open_Once OTC_Loot_Once OTC_AutoWalk_Once OTC_Talk_Once OTC_Look_Watch OTC_Item_Use_Once)
foreach(I RANGE 0 6)
  list(GET SOURCES ${I} SOURCE)
  list(GET TARGETS ${I} TARGET)
  add_executable(${TARGET} src/${SOURCE}.cpp)
  target_link_libraries(${TARGET} PRIVATE user32)
  if(MINGW)
    target_link_options(${TARGET} PRIVATE -static-libgcc -static-libstdc++)
  endif()
endforeach()
